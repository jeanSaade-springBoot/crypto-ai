### DB_REQUEST

QueryId: DBQ-CODEX-I5-003  
Purpose: Reconstruct only contemporaneous 1m evidence plus selected 5m/1h context for the six entries; excludes future signals.  
SQL:
```sql
WITH frozen AS (
    SELECT
        p.id AS paper_position_id,
        p.symbol,
        p.signal_id,
        s.generated_at AS reference_generated_at
    FROM crypto_ai.paper_position p
    JOIN crypto_ai.trade_signal s
      ON s.id = p.signal_id
    WHERE p.id IN (182, 172, 162, 158, 143, 137)
),
context_ranked AS (
    SELECT
        f.paper_position_id,
        ts.id,
        ts.interval_code,
        ts.generated_at,
        ts.candle_open_time,
        ts.decision,
        ts.confidence_score,
        ts.total_score,
        ts.data_quality_valid,
        ROW_NUMBER() OVER (
            PARTITION BY f.paper_position_id, ts.interval_code
            ORDER BY ts.generated_at DESC, ts.id DESC
        ) AS rn
    FROM frozen f
    JOIN crypto_ai.trade_signal ts
      ON ts.symbol = f.symbol
     AND ts.interval_code IN ('5m', '1h')
     AND ts.generated_at <= f.reference_generated_at
)
SELECT
    f.paper_position_id,
    'CONTEXT' AS evidence_type,
    cr.id AS signal_id,
    cr.interval_code,
    cr.generated_at,
    cr.candle_open_time,
    cr.decision,
    cr.confidence_score,
    cr.total_score,
    cr.data_quality_valid
FROM frozen f
JOIN context_ranked cr
  ON cr.paper_position_id = f.paper_position_id
 AND cr.rn = 1

UNION ALL

SELECT
    f.paper_position_id,
    'ONE_MINUTE_WINDOW' AS evidence_type,
    ts.id AS signal_id,
    ts.interval_code,
    ts.generated_at,
    ts.candle_open_time,
    ts.decision,
    ts.confidence_score,
    ts.total_score,
    ts.data_quality_valid
FROM frozen f
JOIN crypto_ai.trade_signal ts
  ON ts.symbol = f.symbol
 AND ts.interval_code = '1m'
 AND ts.generated_at >= f.reference_generated_at - INTERVAL 30 MINUTE
 AND ts.generated_at <= f.reference_generated_at
ORDER BY paper_position_id, evidence_type, generated_at, signal_id;
```
