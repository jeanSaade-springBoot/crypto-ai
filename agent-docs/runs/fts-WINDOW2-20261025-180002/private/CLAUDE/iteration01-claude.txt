### DB_REQUEST
QueryId: DBQ-CLAUDE-I1-004
Purpose: Entry and exit signals behind each position (baseline columns only; extended columns will be requested after I1-001).
SQL:
SELECT p.id AS position_id, 'ENTRY' AS role, s.id AS signal_id, s.symbol, s.interval_code, s.decision, s.total_score, s.trend_score, s.volume_score, s.momentum_score, s.sentiment_score, s.fundamental_score, s.latest_price, s.stop_loss, s.take_profit, s.generated_at, LEFT(s.explanation, 1500) AS explanation_head
FROM crypto_ai.paper_position p JOIN crypto_ai.trade_signal s ON s.id = p.signal_id
WHERE p.id IN (182,172,162,158,143,137)
UNION ALL
SELECT p.id, 'EXIT', s.id, s.symbol, s.interval_code, s.decision, s.total_score, s.trend_score, s.volume_score, s.momentum_score, s.sentiment_score, s.fundamental_score, s.latest_price, s.stop_loss, s.take_profit, s.generated_at, LEFT(s.explanation, 1500)
FROM crypto_ai.paper_position p JOIN crypto_ai.trade_signal s ON s.id = p.exit_signal_id
WHERE p.id IN (182,172,162,158,143,137)
ORDER BY position_id DESC, role
LIMIT 12;
### END_DB_REQUEST

### DB_REQUEST
QueryId: DBQ-CLAUDE-I1-006
Purpose: Wallet fills around each position's lifetime (±2 min), used for P&L and fee reconstruction. The match is by symbol and time window until signal/execution_key confirms it.
SQL:
SELECT p.id AS position_id, w.id AS wallet_trade_id, w.signal_id, w.execution_key, w.side, w.quantity, w.price_usdt, w.gross_amount_usdt, w.fee_usdt, w.net_amount_usdt, w.cost_basis_usdt, w.realized_pnl_usdt, w.realized_pnl_percent, w.execution_type, w.status, w.executed_at, LEFT(w.notes, 300) AS notes_head
FROM crypto_ai.paper_position p
JOIN crypto_ai.wallet_trade w ON w.symbol = p.symbol AND w.executed_at BETWEEN p.opened_at - INTERVAL 2 MINUTE AND p.closed_at + INTERVAL 2 MINUTE
WHERE p.id IN (182,172,162,158,143,137)
ORDER BY p.id DESC, w.executed_at
LIMIT 120;
### END_DB_REQUEST

### DB_REQUEST
QueryId: DBQ-CLAUDE-I1-007
Purpose: How much price-event evidence each position has over its lifetime, and the price range (an aggregate only, before pulling individual rows).
SQL:
SELECT p.id AS position_id, p.symbol, COUNT(e.id) AS events, MIN(e.observed_at) AS first_obs, MAX(e.observed_at) AS last_obs, MIN(e.price) AS min_price, MAX(e.price) AS max_price, MAX(TIMESTAMPDIFF(SECOND, p.opened_at, p.closed_at)) AS held_sec
FROM crypto_ai.paper_position p
LEFT JOIN crypto_ai.market_price_event e ON e.symbol = p.symbol AND e.observed_at BETWEEN p.opened_at AND p.closed_at
WHERE p.id IN (182,172,162,158,143,137)
GROUP BY p.id, p.symbol
ORDER BY p.id DESC
LIMIT 6;
### END_DB_REQUEST

### DB_REQUEST
QueryId: DBQ-CLAUDE-I1-008
Purpose: Managed-position state and management events (SL/TP moves, profit lock) for each position's symbol and lifetime.
SQL:
SELECT p.id AS position_id, m.id AS event_id, m.wallet_position_id, m.event_type, m.old_value_usdt, m.new_value_usdt, m.market_price_usdt, LEFT(m.reason, 400) AS reason_head, m.occurred_at
FROM crypto_ai.paper_position p
JOIN crypto_ai.position_management_event m ON m.symbol = p.symbol AND m.occurred_at BETWEEN p.opened_at - INTERVAL 1 MINUTE AND p.closed_at + INTERVAL 1 MINUTE
WHERE p.id IN (182,172,162,158,143,137)
ORDER BY p.id DESC, m.occurred_at
LIMIT 200;
