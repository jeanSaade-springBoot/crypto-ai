       s.signal_mode, s.decision, s.original_decision, s.total_score, s.confidence_score,

### DB_REQUEST
QueryId: DBQ-CLAUDE-I1-007
Purpose: Get the wallet fills linked to each trade (entry and exit price, decision price versus fill price, price-observation age, quantity, P&L). Match by signal or by symbol inside the position lifetime ±2 min.
SQL:
SELECT pp.id AS paper_position_id, wt.id, wt.signal_id, wt.execution_key, wt.symbol, wt.side,
       wt.quantity, wt.price_usdt, wt.decision_price_usdt, wt.execution_price_observed_at,
       wt.gross_amount_usdt, wt.fee_usdt, wt.net_amount_usdt, wt.cost_basis_usdt,
       wt.realized_pnl_usdt, wt.realized_pnl_percent, wt.execution_type, wt.status,
       wt.executed_at, LEFT(wt.notes, 300) AS notes_300
FROM crypto_ai.paper_position pp
JOIN crypto_ai.wallet_trade wt
  ON wt.symbol = pp.symbol
 AND wt.executed_at BETWEEN pp.opened_at - INTERVAL 2 MINUTE AND pp.closed_at + INTERVAL 2 MINUTE
WHERE pp.id IN (172,162,158,143,137,131)
ORDER BY pp.id, wt.executed_at
LIMIT 200;
### END_DB_REQUEST

### DB_REQUEST
QueryId: DBQ-CLAUDE-I1-008
Purpose: Summarise price-event coverage while each position was open, grouped by delivery status (to find protection gaps).
SQL:
SELECT pp.id AS paper_position_id, e.delivery_status, e.source, COUNT(*) AS events,
       MIN(e.observed_at) AS first_obs, MAX(e.observed_at) AS last_obs,
       MIN(e.price) AS min_price, MAX(e.price) AS max_price,
       MAX(TIMESTAMPDIFF(MICROSECOND, e.observed_at, e.created_at))/1000000 AS max_ingest_lag_s
FROM crypto_ai.paper_position pp
JOIN crypto_ai.market_price_event e
  ON e.symbol = pp.symbol AND e.observed_at BETWEEN pp.opened_at AND pp.closed_at
WHERE pp.id IN (172,162,158,143,137,131)
GROUP BY pp.id, e.delivery_status, e.source
ORDER BY pp.id, e.delivery_status;
### END_DB_REQUEST

### DB_REQUEST
QueryId: DBQ-CLAUDE-I1-009
Purpose: Find the first accepted (LEGACY/APPLIED) price observation at or below the stop-loss for each trade, to compare against the actual close time and price.
SQL:
SELECT pp.id AS paper_position_id, pp.stop_loss, pp.closed_at,
       (SELECT CONCAT(e.id,'|',e.observed_at,'|',e.price,'|',e.delivery_status)
          FROM crypto_ai.market_price_event e
         WHERE e.symbol = pp.symbol
           AND e.observed_at BETWEEN pp.opened_at AND pp.closed_at
           AND e.delivery_status IN ('LEGACY','APPLIED')
           AND e.price <= pp.stop_loss
         ORDER BY e.observed_at, e.id LIMIT 1) AS first_sl_breach
FROM crypto_ai.paper_position pp
WHERE pp.id IN (172,162,158,143,137,131)
ORDER BY pp.id;
### END_DB_REQUEST

### DB_REQUEST
QueryId: DBQ-CLAUDE-I1-010
Purpose: Get the candle each entry signal was built from (closed flag, provenance, close token) to check data freshness and integrity.
SQL:
SELECT pp.id AS paper_position_id, c.*
FROM crypto_ai.paper_position pp
JOIN crypto_ai.trade_signal s ON s.id = pp.signal_id
JOIN crypto_ai.candle c
  ON c.symbol = s.symbol AND c.interval_code = s.interval_code AND c.open_time = s.candle_open_time
WHERE pp.id IN (172,162,158,143,137,131)
ORDER BY pp.id;
