### DB_REQUEST
QueryId: DBQ-CLAUDE-I1-005
Purpose: Wallet executions inside each position's lifetime (entry, adds, partial harvests, exit), for fill authority, fill delay, and P&L recomputation.
SQL:
SELECT p.id AS position_id, w.id AS wallet_trade_id, w.signal_id, w.execution_key,
       w.side, w.quantity, w.price_usdt, w.decision_price_usdt,
       w.execution_price_observed_at, w.executed_at,
       TIMESTAMPDIFF(MICROSECOND, w.execution_price_observed_at, w.executed_at)/1e6 AS fill_age_s,
       w.gross_amount_usdt, w.fee_usdt, w.net_amount_usdt, w.cost_basis_usdt,
       w.realized_pnl_usdt, w.realized_pnl_percent, w.execution_type,
       w.execution_reason, w.status, LEFT(w.notes, 300) AS notes_head
FROM paper_position p
JOIN wallet_trade w
  ON w.symbol = p.symbol
 AND w.executed_at BETWEEN p.opened_at - INTERVAL 2 MINUTE AND p.closed_at + INTERVAL 2 MINUTE
WHERE p.id IN (182,172,162,158,143,137)
ORDER BY p.id DESC, w.executed_at
LIMIT 200;
### END_DB_REQUEST

### DB_REQUEST
QueryId: DBQ-CLAUDE-I1-007
Purpose: Summary of price observations during each position's lifetime (count, range, first/last, biggest gap between observations) to find protection gaps and the max favourable/adverse move. Per-row detail comes in iteration 2.
SQL:
SELECT p.id AS position_id, p.symbol, COUNT(*) AS events,
       MIN(e.price) AS min_price, MAX(e.price) AS max_price,
       MIN(e.observed_at) AS first_obs, MAX(e.observed_at) AS last_obs,
       (MAX(e.price) - p.entry_price) / p.entry_price * 100 AS mfe_pct,
       (MIN(e.price) - p.entry_price) / p.entry_price * 100 AS mae_pct
FROM paper_position p
JOIN market_price_event e
  ON e.symbol = p.symbol
 AND e.observed_at BETWEEN p.opened_at AND p.closed_at
WHERE p.id IN (182,172,162,158,143,137)
GROUP BY p.id, p.symbol, p.entry_price
ORDER BY p.id DESC;
### END_DB_REQUEST

### DB_REQUEST
QueryId: DBQ-CLAUDE-I1-008
Purpose: Freshness and admission of each entry signal: the persisted 1m candle and its provenance/closed flag, compared with the signal time.
SQL:
SELECT p.id AS position_id, c.id AS candle_id, c.symbol, c.interval_code,
       c.open_time, c.close_time, c.closed, c.open_price, c.high_price,
       c.low_price, c.close_price, c.volume, c.created_at,
       s.generated_at,
       TIMESTAMPDIFF(MICROSECOND, c.close_time, s.generated_at)/1e6 AS close_to_signal_s,
       TIMESTAMPDIFF(MICROSECOND, s.generated_at, p.opened_at)/1e6 AS signal_to_open_s
FROM paper_position p
JOIN trade_signal s ON s.id = p.signal_id
JOIN candle c ON c.symbol = s.symbol AND c.interval_code = s.interval_code
             AND c.open_time = s.candle_open_time
WHERE p.id IN (182,172,162,158,143,137)
ORDER BY p.id DESC;
### END_DB_REQUEST
