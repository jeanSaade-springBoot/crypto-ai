```sql
SHOW CREATE TABLE crypto_ai.paper_position;
SHOW CREATE TABLE crypto_ai.wallet_trade;
SHOW CREATE TABLE crypto_ai.wallet_managed_position;
SHOW CREATE TABLE crypto_ai.production_exit_audit;
SHOW CREATE TABLE crypto_ai.position_analysis;
SHOW CREATE TABLE crypto_ai.market_price_event;

SELECT
    p.id, p.symbol, p.side, p.status, p.quantity,
    p.entry_price, p.stop_loss, p.take_profit, p.exit_price, p.realized_pnl,
    p.signal_id, p.exit_signal_id, p.close_reason, p.entry_reason, p.exit_reason,
    p.opened_at, p.closed_at
FROM crypto_ai.paper_position p
WHERE p.id IN (182, 172, 162, 158, 143, 137)
ORDER BY FIELD(p.id, 182, 172, 162, 158, 143, 137);

SELECT
    a.*
FROM crypto_ai.production_exit_audit a
WHERE a.paper_position_id IN (182, 172, 162, 158, 143, 137)
ORDER BY a.paper_position_id, a.audited_at, a.id;

SELECT
    p.id AS paper_position_id,
    m.*
FROM crypto_ai.paper_position p
JOIN crypto_ai.wallet_managed_position m
  ON m.entry_signal_id = p.signal_id
WHERE p.id IN (182, 172, 162, 158, 143, 137)
ORDER BY p.id, m.opened_at, m.id;

SELECT
    p.id AS paper_position_id,
    w.*
FROM crypto_ai.paper_position p
LEFT JOIN crypto_ai.production_exit_audit a
  ON a.paper_position_id = p.id
JOIN crypto_ai.wallet_trade w
  ON w.signal_id = p.signal_id
  OR w.signal_id = p.exit_signal_id
  OR (
      a.wallet_position_id IS NOT NULL
      AND w.execution_key LIKE CONCAT('POSITION:', a.wallet_position_id, ':%')
  )
  OR (
      a.position_analysis_id IS NOT NULL
      AND w.execution_key = CONCAT('POSITION_ANALYSIS:', a.position_analysis_id, ':STOP_LOSS')
  )
WHERE p.id IN (182, 172, 162, 158, 143, 137)
ORDER BY p.id, w.executed_at, w.id;

SELECT
    refs.paper_position_id,
    refs.signal_role,
    s.id, s.symbol, s.interval_code, s.candle_open_time,
    s.decision, s.original_decision, s.final_entry_allowed,
    s.latest_price, s.stop_loss, s.take_profit,
    s.total_score, s.confidence_score, s.risk_reward_ratio,
    s.data_quality_valid, s.signal_mode, s.generated_at,
    s.explanation, s.final_decision_explanation, s.decision_path
FROM (
    SELECT p.id AS paper_position_id, 'ENTRY' AS signal_role, p.signal_id AS signal_id
    FROM crypto_ai.paper_position p
    WHERE p.id IN (182, 172, 162, 158, 143, 137)

    UNION ALL

    SELECT p.id, 'PAPER_EXIT_SIGNAL', p.exit_signal_id
    FROM crypto_ai.paper_position p
    WHERE p.id IN (182, 172, 162, 158, 143, 137)
      AND p.exit_signal_id IS NOT NULL

    UNION ALL

    SELECT a.paper_position_id, 'LIVE_EXIT_CONTEXT', a.source_signal_id
    FROM crypto_ai.production_exit_audit a
    WHERE a.paper_position_id IN (182, 172, 162, 158, 143, 137)
      AND a.source_signal_id IS NOT NULL
) refs
JOIN crypto_ai.trade_signal s ON s.id = refs.signal_id
ORDER BY refs.paper_position_id, refs.signal_role, s.id;

SELECT
    p.id AS paper_position_id,
    pa.*
FROM crypto_ai.paper_position p
JOIN crypto_ai.production_exit_audit a
  ON a.paper_position_id = p.id
JOIN crypto_ai.position_analysis pa
  ON pa.wallet_position_id = a.wallet_position_id
WHERE p.id IN (182, 172, 162, 158, 143, 137)
ORDER BY p.id, pa.analyzed_at, pa.id;

SELECT

    w.*
FROM crypto_ai.paper_position p
JOIN crypto_ai.signal_processing_work w
  ON w.signal_id = p.signal_id
  OR w.signal_id = p.exit_signal_id
WHERE p.id IN (182, 172, 162, 158, 143, 137)
ORDER BY p.id, w.signal_id, w.id;

SELECT
    p.id AS paper_position_id,
    e.id, e.symbol, e.observed_at, e.price, e.source,
    e.delivery_status, e.ordering_slot, e.outcome_reason, e.created_at
FROM crypto_ai.paper_position p
JOIN crypto_ai.market_price_event e
  ON e.symbol = p.symbol
 AND e.observed_at >= DATE_SUB(p.closed_at, INTERVAL 15 MINUTE)
 AND e.observed_at <= DATE_ADD(p.closed_at, INTERVAL 2 MINUTE)
WHERE p.id IN (182, 172, 162, 158, 143, 137)
```
