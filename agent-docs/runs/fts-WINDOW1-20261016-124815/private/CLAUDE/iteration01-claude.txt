### DB_REQUEST
QueryId: DBQ-CLAUDE-I1-004
Purpose: Get the entry and exit signal decision chains (columns confirmed in the baseline migrations; large JSON excluded).
SQL:
SELECT pp.id AS position_id,
       CASE WHEN ts.id = pp.signal_id THEN 'ENTRY' ELSE 'EXIT' END AS role,
       ts.id AS signal_id, ts.symbol, ts.interval_code, ts.candle_open_time, ts.generated_at, ts.created_at,
       ts.decision, ts.original_decision, ts.total_score, ts.confidence_score, ts.raw_confidence_score,
       ts.effective_confidence_score, ts.final_entry_allowed, ts.primary_blocking_stage, ts.entry_authority,
       ts.latest_price, ts.stop_loss, ts.take_profit, ts.risk_reward_ratio, ts.atr_percent, ts.atr_entry_type,
       ts.atr_recommended_position_percent, ts.market_regime, ts.market_regime_confidence, ts.selected_strategy,

### DB_REQUEST

### DB_REQUEST
QueryId: DBQ-CLAUDE-I1-006
Purpose: Get the wallet executions (BUY, partial harvests, SELL) inside each position's lifetime, for paper-vs-wallet P&L and fill-price checks.
SQL:
SELECT pp.id AS position_id, wt.id AS wallet_trade_id, wt.signal_id, wt.execution_key, wt.position_analysis_id,
       wt.side, wt.quantity, wt.price_usdt, wt.decision_price_usdt, wt.execution_price_observed_at,
       wt.gross_amount_usdt, wt.fee_usdt, wt.net_amount_usdt, wt.cost_basis_usdt, wt.realized_pnl_usdt,
       wt.realized_pnl_percent, wt.execution_type, wt.execution_reason, wt.status, wt.executed_at,
       LEFT(wt.execution_message, 300) AS execution_message_head
FROM crypto_ai.paper_position pp
JOIN crypto_ai.wallet_trade wt
  ON wt.symbol = pp.symbol
 AND wt.executed_at BETWEEN pp.opened_at - INTERVAL 60 SECOND AND pp.closed_at + INTERVAL 60 SECOND
WHERE pp.id IN (182,172,162,158,143,137)
ORDER BY pp.id DESC, wt.executed_at
LIMIT 200;
### END_DB_REQUEST

### DB_REQUEST
QueryId: DBQ-CLAUDE-I1-007
Purpose: Measure price-event coverage per position lifetime by delivery status. This finds protection gaps (PENDING / NON_CANONICAL / PROTECTION_NOT_APPLIED) and the extreme prices.
SQL:
SELECT pp.id AS position_id, pp.symbol, mpe.delivery_status, COUNT(*) AS events,
       MIN(mpe.observed_at) AS first_obs, MAX(mpe.observed_at) AS last_obs,
       MIN(mpe.price) AS min_price, MAX(mpe.price) AS max_price
FROM crypto_ai.paper_position pp
JOIN crypto_ai.market_price_event mpe
  ON mpe.symbol = pp.symbol
 AND mpe.observed_at BETWEEN pp.opened_at - INTERVAL 30 SECOND AND pp.closed_at + INTERVAL 30 SECOND
WHERE pp.id IN (182,172,162,158,143,137)
GROUP BY pp.id, pp.symbol, mpe.delivery_status
ORDER BY pp.id DESC, mpe.delivery_status;
### END_DB_REQUEST
