### DB_REQUEST

QueryId: DBQ-CODEX-I4-001  
Purpose: Prove each frozen paper position’s durable route, including entry/exit signals, managed-position linkage, recorded terminal trigger, wallet execution evidence, and accepted/non-canonical price events around close. All timestamps returned are UTC.  
SQL:

```sql
SELECT
    pp.id AS paper_position_id,
    pp.symbol,
    pp.status,
    pp.quantity,
    pp.entry_price,
    pp.stop_loss,
    pp.take_profit,
    pp.exit_price,
    pp.realized_pnl,
    pp.signal_id AS entry_signal_id,
    pp.exit_signal_id,
    pp.close_reason,
    pp.exit_reason,
    pp.opened_at,
    pp.closed_at,
    es.interval_code AS entry_interval,
    es.decision AS entry_decision,
    es.original_decision AS entry_original_decision,
    es.latest_price AS entry_signal_price,
    es.generated_at AS entry_signal_generated_at,
    xs.interval_code AS exit_interval,
    xs.decision AS exit_decision,
    xs.original_decision AS exit_original_decision,
    xs.latest_price AS exit_signal_price,
    xs.generated_at AS exit_signal_generated_at,
    wmp.id AS wallet_managed_position_id,
    wmp.status AS wallet_managed_position_status,
    wmp.average_entry_price_usdt,
    wmp.stop_loss_usdt,
    wmp.take_profit_usdt,
    pea.id AS production_exit_audit_id,
    pea.close_trigger AS audit_close_trigger,
    pea.source_signal_id AS audit_source_signal_id,
    pea.source_signal_decision AS audit_source_signal_decision,
    pea.wallet_position_id AS audit_wallet_position_id,
    pea.close_explanation AS audit_close_explanation,
    pea.audited_at
FROM paper_position pp
LEFT JOIN trade_signal es ON es.id = pp.signal_id
LEFT JOIN trade_signal xs ON xs.id = pp.exit_signal_id
LEFT JOIN wallet_managed_position wmp ON wmp.entry_signal_id = pp.signal_id
LEFT JOIN production_exit_audit pea ON pea.paper_position_id = pp.id
WHERE pp.id IN (182, 172, 162, 158, 143, 137)
ORDER BY FIELD(pp.id, 182, 172, 162, 158, 143, 137);

SELECT
    pp.id AS paper_position_id,
    wmp.id AS wallet_managed_position_id,
    wt.id AS wallet_trade_id,
    wt.signal_id,
    wt.execution_key,
    wt.symbol,
    wt.side,
    wt.quantity,
    wt.price_usdt,
    wt.decision_price_usdt,
    wt.execution_price_observed_at,
    wt.gross_amount_usdt,
    wt.fee_usdt,
    wt.net_amount_usdt,
    wt.cost_basis_usdt,
    wt.realized_pnl_usdt,
    wt.execution_reason,
    wt.status,
    wt.executed_at,
    wt.execution_message
FROM paper_position pp
LEFT JOIN wallet_managed_position wmp ON wmp.entry_signal_id = pp.signal_id
JOIN wallet_trade wt
  ON wt.signal_id IN (pp.signal_id, pp.exit_signal_id)
  OR (
      wmp.id IS NOT NULL
      AND wt.execution_key LIKE CONCAT('POSITION:', wmp.id, ':%')
  )
WHERE pp.id IN (182, 172, 162, 158, 143, 137)
ORDER BY pp.id, wt.executed_at, wt.id;

SELECT
    pp.id AS paper_position_id,
    pp.symbol,
    pp.closed_at,
    mpe.id AS market_price_event_id,
    mpe.observed_at,
    mpe.price,
    mpe.source,
    mpe.delivery_status,
    mpe.ordering_slot,
    mpe.outcome_reason,
    mpe.created_at
FROM paper_position pp
JOIN market_price_event mpe
  ON mpe.symbol = pp.symbol
 AND mpe.observed_at BETWEEN DATE_SUB(pp.closed_at, INTERVAL 15 SECOND)
                         AND DATE_ADD(pp.closed_at, INTERVAL 15 SECOND)
WHERE pp.id IN (182, 172, 162, 158, 143, 137)
ORDER BY pp.id, mpe.observed_at, mpe.id;
```
