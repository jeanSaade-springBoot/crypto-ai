```sql

    table_name,
    ordinal_position,
    column_name,
    column_type,
    is_nullable,
    column_key,
    extra

WHERE table_schema = 'crypto_ai'

      'wallet_managed_position',
      'market_price_event',

      'production_exit_audit',
      'candle',
      'technical_indicator',
      'candle_analysis_outcome'
```
