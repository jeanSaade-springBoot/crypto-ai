WITH frozen AS (

            PARTITION BY f.paper_position_id
            ORDER BY mpe.observed_at DESC, mpe.id DESC
        ) AS recency_rank
    FROM frozen f
    JOIN crypto_ai.market_price_event mpe
      ON mpe.symbol = f.symbol
     AND mpe.observed_at >= f.closed_at - INTERVAL 180 SECOND
     AND mpe.observed_at <= f.closed_at + INTERVAL 1 SECOND
)
SELECT *
FROM ranked_events
WHERE recency_rank <= 200
ORDER BY paper_position_id, observed_at, market_price_event_id;
