### DB_REQUEST
QueryId: DBQ-CODEX-I7-001  
Purpose: Independently verify the exact frozen selection as-of the stated UTC snapshot and retrieve the minimum position-linked exit evidence required to avoid falsely attributing symbol-level price events to any of the six frozen positions. Returned timestamps are UTC.

SQL:
```sql
WITH eligible AS (
    SELECT
        p.id,
        p.symbol,
        p.status,
        p.closed_at,
        p.realized_pnl
    FROM crypto_ai.paper_position p
    WHERE p.status = 'CLOSED'
      AND p.closed_at IS NOT NULL
      AND p.realized_pnl < 0
      AND p.closed_at <= TIMESTAMP('2026-10-12 15:00:04')
)
SELECT
    id,
    symbol,
    status,
    closed_at,
    realized_pnl
FROM eligible
ORDER BY closed_at DESC
LIMIT 6;

SELECT
    p.id AS paper_position_id,
    p.symbol,
    p.side,
    p.status,
    p.quantity,
    p.entry_price,
    p.stop_loss,
    p.take_profit,
    p.exit_price,
    p.realized_pnl,
    p.close_reason,
    p.signal_id,
    p.exit_signal_id,
    p.opened_at,
    p.closed_at,
    pea.id AS production_exit_audit_id,
    pea.wallet_position_id,
    pea.close_trigger,
    pea.source_signal_id,
    pea.source_signal_decision,
    pea.source_signal_original_decision,
    pea.position_analysis_id,
    pea.position_recommendation,
    pea.entry_price_usdt AS audited_entry_price,
    pea.exit_price_usdt AS audited_exit_price,
    pea.stop_loss_usdt AS audited_stop_loss,
    pea.take_profit_usdt AS audited_take_profit,
    pea.close_explanation,
    pea.audited_at
FROM crypto_ai.paper_position p
LEFT JOIN crypto_ai.production_exit_audit pea
    ON pea.paper_position_id = p.id
WHERE p.id IN (182, 172, 162, 158, 143, 137)
ORDER BY p.closed_at DESC, p.id DESC;

SELECT
    pea.paper_position_id,
    pme.id AS management_event_id,
    pme.wallet_position_id,
    pme.symbol,
    pme.event_type,
    pme.old_value_usdt,
    pme.new_value_usdt,
    pme.market_price_usdt,
    pme.reason,
    pme.occurred_at
FROM crypto_ai.production_exit_audit pea
JOIN crypto_ai.position_management_event pme
    ON pme.wallet_position_id = pea.wallet_position_id
WHERE pea.paper_position_id IN (182, 172, 162, 158, 143, 137)
ORDER BY pea.paper_position_id, pme.occurred_at, pme.id;
```
