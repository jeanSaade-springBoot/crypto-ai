QueryId: DBQ-CODEX-I5-001  
Purpose: Establish the actual snapshot schema and keys before requesting entry-quality evidence for the six frozen `paper_position` IDs. This preserves the requirement to verify physical columns/relationships before querying trade rows.  
SQL:
```sql
SELECT
    c.table_name,
    c.ordinal_position,
    c.column_name,
    c.column_type,
    c.is_nullable,
    c.column_key,
    c.column_default,
    c.extra,
    k.constraint_name,
    k.referenced_table_name,
    k.referenced_column_name
FROM information_schema.columns AS c
LEFT JOIN information_schema.key_column_usage AS k
    ON k.table_schema = c.table_schema
   AND k.table_name = c.table_name
   AND k.column_name = c.column_name
   AND k.referenced_table_name IS NOT NULL
WHERE c.table_schema = 'crypto_ai'
  AND c.table_name IN (
      'paper_position',
      'trade_signal',
      'wallet_trade',
      'wallet_managed_position',
      'signal_processing_work',
      'candle',
      'technical_indicator'
  )
ORDER BY c.table_name, c.ordinal_position, k.constraint_name;

SELECT
    table_name,
    index_name,
    non_unique,
    seq_in_index,
    column_name
FROM information_schema.statistics
WHERE table_schema = 'crypto_ai'
  AND table_name IN (
      'paper_position',
      'trade_signal',
      'wallet_trade',
      'wallet_managed_position',
      'signal_processing_work',
      'candle',
      'technical_indicator'
  )
ORDER BY table_name, index_name, seq_in_index;
```
