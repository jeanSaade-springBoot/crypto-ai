```sql
SELECT
    table_name,
    ordinal_position,
    column_name,
    column_type,
    is_nullable,
    column_key

  AND table_name IN (
      'paper_position',
      'trade_signal',
      'wallet_trade',
      'wallet_managed_position',
      'signal_processing_work',
      'market_price_event'

FROM crypto_ai.paper_position AS p
```
